    54→/** Compact UTC minute stamp for panel rows, e.g. "2026-11-14 08:05Z". */
    55→function utcMin(iso: string): string {
    56→  const t = Date.parse(iso);
    57→  if (!Number.isFinite(t)) return '—';
    58→  return `${new Date(t).toISOString().slice(0, 16).replace('T', ' ')}Z`;
    59→}

   110→/** Compact per-day detection bars (real counts from the last evaluation). */
   111→function DayBars({ perDay }: { perDay: { date: string; total: number; qualifying: number }[] }) {
   112→  const max = Math.max(1, ...perDay.map((d) => d.total));
   113→  return (
   114→    <div className="flex items-end gap-1" role="img" aria-label="Detections per day in the evaluation window">
   115→      {perDay.map((d) => {
   116→        const h = Math.max(2, Math.round((d.total / max) * 28));
   117→        const qh = Math.max(0, Math.round((d.qualifying / max) * 28));
   118→        return (
   119→          <div key={d.date} className="flex flex-col items-center gap-1" title={`${d.date}: ${d.total} detection(s), ${d.qualifying} qualifying`}>
   120→            <div
   121→              className="relative w-4 overflow-hidden rounded-sm border border-edge bg-panel-2/60"
   122→              style={{ height: 30 }}
   123→              aria-hidden
   124→            >
   125→              <div className="absolute bottom-0 w-full bg-dim/40" style={{ height: h }} />
   126→              <div className="absolute bottom-0 w-full bg-caution" style={{ height: Math.min(qh, h) }} />
   127→            </div>
   128→            <span className="telemetry text-[8px] text-faint">{d.date.slice(8)}</span>
   129→          </div>
   130→        );
   131→      })}
   132→    </div>
   133→  );
   134→}
