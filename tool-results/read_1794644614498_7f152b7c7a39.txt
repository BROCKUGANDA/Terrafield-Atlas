    60→/** Compact per-day detection bars (real counts from the last evaluation). */
    61→function DayBars({ perDay }: { perDay: { date: string; total: number; qualifying: number }[] }) {
    62→  const max = Math.max(1, ...perDay.map((d) => d.total));
    63→  return (
    64→    <div className="flex items-end gap-1" role="img" aria-label="Detections per day in the evaluation window">
    65→      {perDay.map((d) => {
    66→        const h = Math.max(2, Math.round((d.total / max) * 28));
    67→        const qh = Math.max(0, Math.round((d.qualifying / max) * 28));
    68→        return (
    69→          <div key={d.date} className="flex flex-col items-center gap-1" title={`${d.date}: ${d.total} detection(s), ${d.qualifying} qualifying`}>
    70→            <div
    71→              className="relative w-4 overflow-hidden rounded-sm border border-edge bg-panel-2/60"
    72→              style={{ height: 30 }}
    73→              aria-hidden
    74→            >
    75→              <div className="absolute bottom-0 w-full bg-dim/40" style={{ height: h }} />
    76→              <div className="absolute bottom-0 w-full bg-caution" style={{ height: Math.min(qh, h) }} />
    77→            </div>
    78→            <span className="telemetry text-[8px] text-faint">{d.date.slice(8)}</span>
    79→          </div>
    80→        );
    81→      })}
    82→    </div>
    83→  );
    84→}

   452→              editing={editingId === w.id}
   453→              editForm={editForm}

   475→                  category: 'FIRE',
   476→                  title: `Console geography set — ${w.label ?? w.name}`,
   477→                  detail: `Targeted from fire watchlist “${w.name}” (${w.lat.toFixed(2)}°, ${w.lon.toFixed(2)}°, radius ${w.radiusKm.toFixed(0)} km).`,

   485→            />

   487→        </div>

   489→        <p className="text-[10px] leading-relaxed text-faint">
   490→          Watchlist states describe the last recorded evaluation only. A header alert chip appears console-wide while any

   492→          VIIRS observation with provenance in Evidence &amp; Provenance.
   493→        </p>
   494→      </HudPanel>
   495→    </motion.section>
