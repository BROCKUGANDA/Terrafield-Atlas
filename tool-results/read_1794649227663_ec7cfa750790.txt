    63→/** Compact per-day detection bars (real counts from the last evaluation). */
    64→function DayBars({ perDay }: { perDay: { date: string; total: number; qualifying: number }[] }) {
    65→  const max = Math.max(1, ...perDay.map((d) => d.total));
    66→  return (
    67→    <div className="flex items-end gap-1" role="img" aria-label="Detections per day in the evaluation window">
    68→      {perDay.map((d) => {
    69→        const h = Math.max(2, Math.round((d.total / max) * 28));
    70→        const qh = Math.max(0, Math.round((d.qualifying / max) * 28));
    71→        return (
    72→          <div key={d.date} className="flex flex-col items-center gap-1" title={`${d.date}: ${d.total} detection(s), ${d.qualifying} qualifying`}>
    73→            <div
    74→              className="relative w-4 overflow-hidden rounded-sm border border-edge bg-panel-2/60"
    75→              style={{ height: 30 }}
    76→              aria-hidden
    77→            >
    78→              <div className="absolute bottom-0 w-full bg-dim/40" style={{ height: h }} />
    79→              <div className="absolute bottom-0 w-full bg-caution" style={{ height: Math.min(qh, h) }} />
    80→            </div>
    81→            <span className="telemetry text-[8px] text-faint">{d.date.slice(8)}</span>
    82→          </div>
    83→        );
    84→      })}
    85→    </div>
    86→  );
    87→}
