   301→function CropHistoryEditor({ farmId, current }: { farmId: string; current: string | null }) {
   302→  const { toast } = useToast();
   303→  const queryClient = useQueryClient();
   304→  const [entries, setEntries] = useState<{ season: string; crop: string; family?: string }[]>(
   305→    current ? JSON.parse(current) : [],
   306→  );
   307→  const update = useMutation({
   308→    mutationFn: () => api.updateFarm(farmId, { cropHistory: entries }),
   309→    onSuccess: () => {
   310→      queryClient.invalidateQueries({ queryKey: ['farm'] });
   311→      toast({ title: 'Crop history saved' });
   312→    },
   313→  });
   314→  return (
   315→    <div className="mt-4 rounded-sm border border-edge bg-panel-2/40 p-3">
   316→      <div className="flex items-center gap-1.5 text-xs font-medium text-dim">
   317→        <Sprout className="h-3.5 w-3.5 text-signal" aria-hidden /> Previous crop sequence (farmer-supplied)
   318→      </div>
   319→      <div className="mt-2 flex flex-col gap-2">
   320→        {entries.map((e, i) => (
   321→          <div key={i} className="flex flex-wrap items-center gap-2 text-xs">
   322→            <Input value={e.season} onChange={(ev) => setEntries(entries.map((x, j) => j === i ? { ...x, season: ev.target.value } : x))} placeholder="Season (e.g. 2025 long rains)" className="h-8 w-44 border-edge bg-panel text-xs text-ink" aria-label={`Season ${i + 1}`} />
   323→            <Input value={e.crop} onChange={(ev) => setEntries(entries.map((x, j) => j === i ? { ...x, crop: ev.target.value } : x))} placeholder="Crop" className="h-8 w-36 border-edge bg-panel text-xs text-ink" aria-label={`Crop ${i + 1}`} />
   324→            <Button variant="ghost" size="sm" className="h-8 px-2 text-faint hover:text-crit" onClick={() => setEntries(entries.filter((_, j) => j !== i))} aria-label={`Remove entry ${i + 1}`}>Remove</Button>
   325→          </div>
   326→        ))}
   327→        <div className="flex gap-2">
   328→          <Button variant="outline" size="sm" className={cn('h-8 uppercase', MONO_BTN)} onClick={() => setEntries([...entries, { season: '', crop: '' }])}>Add season</Button>
   329→          <Button size="sm" className={cn('h-8 uppercase', MONO_BTN)} disabled={update.isPending} onClick={() => update.mutate()}>
   330→            {update.isPending ? (
   331→              <>
   332→                <span className="anim-blink h-1.5 w-1.5 rounded-full bg-primary-foreground" aria-hidden /> Saving history…
   333→              </>
   334→            ) : (
   335→              'Save history'
   336→            )}
   337→          </Button>
   338→        </div>
   339→      </div>
   340→    </div>
   341→  );
   342→}

   350→function SoilSection({ farmId, tests, onSoilGrids, soilGridsPending }: { farmId: string; tests: SoilTest[]; onSoilGrids: () => void; soilGridsPending: boolean }) {
   351→  const { toast } = useToast();
   352→  const queryClient = useQueryClient();
   353→  const [form, setForm] = useState({
   354→    source: 'farmer' as 'farmer' | 'laboratory',
   355→    measuredAt: new Date().toISOString().slice(0, 10),
   356→    ph: '', texture: '', organicCarbonPct: '', nitrogenMgKg: '', phosphorusMgKg: '', potassiumCmolKg: '',
   357→    sandPct: '', siltPct: '', clayPct: '', depthCm: '', notes: '',
   358→  });
   359→  const [textureSelect, setTextureSelect] = useState('none');
   360→
   361→  const addTest = useMutation({
   362→    mutationFn: () => {
   363→      const num = (v: string) => (v.trim() === '' ? undefined : Number(v));
   364→      return api.addSoilTest(farmId, {
   365→        source: form.source,
   366→        measuredAt: form.measuredAt,
   367→        ph: num(form.ph),
   368→        texture: textureSelect !== 'none' ? textureSelect : undefined,
   369→        organicCarbonPct: num(form.organicCarbonPct),
   370→        nitrogenMgKg: num(form.nitrogenMgKg),
   371→        phosphorusMgKg: num(form.phosphorusMgKg),
   372→        potassiumCmolKg: num(form.potassiumCmolKg),
   373→        sandPct: num(form.sandPct),
   374→        siltPct: num(form.siltPct),
   375→        clayPct: num(form.clayPct),
   376→        depthCm: num(form.depthCm),
   377→        notes: form.notes.trim() || undefined,
   378→      });
   379→    },
   380→    onSuccess: () => {
   381→      queryClient.invalidateQueries({ queryKey: ['farm'] });
   382→      toast({ title: 'Soil measurement stored', description: `Labeled as ${form.source} measurement — never mixed with modeled values.` });
   383→      setForm({ ...form, ph: '', organicCarbonPct: '', nitrogenMgKg: '', phosphorusMgKg: '', potassiumCmolKg: '', sandPct: '', siltPct: '', clayPct: '', notes: '' });
   384→    },
   385→    onError: (err) => toast({ title: 'Validation failed', description: err instanceof ApiRequestError ? err.message : 'check values', variant: 'destructive' }),
   386→  });
   387→
   388→  return (
   389→    <HudPanel>
   390→      <div className="flex items-center gap-2">
   391→        <FlaskConical className="h-4 w-4 text-caution" aria-hidden />
   392→        <h3 className="font-display text-sm font-bold uppercase tracking-wide-mc text-ink">Soil properties</h3>
   393→      </div>
   394→
   395→      {/* Source hierarchy note */}
   396→      <div className="mt-3 flex items-start gap-2.5 rounded-sm border border-caution/35 bg-caution/5 p-3">
   397→        <AlertTriangle className="mt-0.5 h-3.5 w-3.5 shrink-0 text-caution" aria-hidden />
   398→        <p className="text-[11px] leading-relaxed text-dim">
   399→          <span className="font-mono uppercase tracking-wide-mc text-caution">Source hierarchy</span> — farmer / laboratory measurements
   400→          always outrank the modeled SoilGrids supplementary profile. Modeled values are clearly labeled model
   401→          estimates and are never mixed with measurements.
   402→        </p>
   403→      </div>

   530→        <Button size="sm" className={cn('mt-3 h-8 uppercase', MONO_BTN)} disabled={addTest.isPending} onClick={() => addTest.mutate()}>

   537→          )}
   538→        </Button>
   539→      </div>
   540→    </HudPanel>
   541→  );
   542→}
