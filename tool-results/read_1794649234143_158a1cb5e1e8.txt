    64→    64→function DayBars({ perDay }: { perDay: { date: string; total: number; qualifying: number }[] }) {
    65→    65→  const max = Math.max(1, ...perDay.map((d) => d.total));
    66→    66→  return (
    67→    67→    <div className="flex items-end gap-1" role="img" aria-label="Detections per day in the evaluation window">
    68→    68→      {perDay.map((d) => {
    69→    69→        const h = Math.max(2, Math.round((d.total / max) * 28));
    70→    70→        const qh = Math.max(0, Math.round((d.qualifying / max) * 28));
    71→    71→        return (
    72→    72→          <div key={d.date} className="flex flex-col items-center gap-1" title={`${d.date}: ${d.total} detection(s), ${d.qualifying} qualifying`}>
    73→    73→            <div
    74→    74→              className="relative w-4 overflow-hidden rounded-sm border border-edge bg-panel-2/60"
    75→    75→              style={{ height: 30 }}
    76→    76→              aria-hidden
    77→    77→            >
    78→    78→              <div className="absolute bottom-0 w-full bg-dim/40" style={{ height: h }} />
    79→    79→              <div className="absolute bottom-0 w-full bg-caution" style={{ height: Math.min(qh, h) }} />
    80→    80→            </div>
    81→    81→            <span className="telemetry text-[8px] text-faint">{d.date.slice(8)}</span>
    82→    82→          </div>
    83→    83→        );
    84→    84→      })}
    85→    85→    </div>
    86→    86→  );
    87→    87→}
